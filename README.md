```rust
/// Example of my documentation style in Rust
/// 
/// This function calculates the square root of a number,
/// returning an error for negative inputs.
/// 
/// # Arguments
/// 
/// * `number` - A floating-point value to find the square root of
/// 
/// # Returns
/// 
/// * `Ok(f64)` - The square root of the input
/// * `Err(String)` - An error message if the input is negative
/// 
/// # Examples
/// 
/// ```
/// let result = sqrt(4.0)?;
/// assert_eq!(result, 2.0);
/// ```
fn sqrt(number: f64) -> Result<f64, String> {
    if number >= 0.0 {
        Ok(number.powf(0.5))
    } else {
        Err("negative floats don't have square roots".to_owned())
    }
}
```
